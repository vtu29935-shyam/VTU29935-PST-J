import java.util.*;

public class Main {

    public static boolean isLapindrome(String s) {
        int n = s.length();

        // Ignore the middle character if length is odd
        int mid = n / 2;

        int[] left = new int[26];
        int[] right = new int[26];

        // Count characters in the left half
        for (int i = 0; i < mid; i++) {
            left[s.charAt(i) - 'a']++;
        }

        // Count characters in the right half
        for (int i = (n + 1) / 2; i < n; i++) {
            right[s.charAt(i) - 'a']++;
        }

        // Compare frequencies
        return Arrays.equals(left, right);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int T = sc.nextInt();

        while (T-- > 0) {
            String s = sc.next();
            
            if (isLapindrome(s)) {
                System.out.println("YES");
            } else {
                System.out.println("NO");
            }
        }

        sc.close();
    }
}
