import java.io.*;
import java.util.*;

public class Main {

    static int[] radius;

    // Manacher's algorithm
    static void manacher(char[] a) {
        int n = a.length;
        radius = new int[n];

        int center = 0;
        int right = 0;

        for (int i = 0; i < n; i++) {

            if (i < right) {
                radius[i] = Math.min(
                        right - i,
                        radius[2 * center - i]
                );
            }

            while (i - radius[i] - 1 >= 0 &&
                   i + radius[i] + 1 < n &&
                   a[i - radius[i] - 1] == a[i + radius[i] + 1]) {
                radius[i]++;
            }

            if (i + radius[i] > right) {
                center = i;
                right = i + radius[i];
            }
        }
    }

    // Sparse table for range maximum query
    static class SparseTable {

        int[][] table;
        int[] log;

        SparseTable(int[] arr) {
            int n = arr.length;

            log = new int[n + 1];

            for (int i = 2; i <= n; i++) {
                log[i] = log[i / 2] + 1;
            }

            int levels = log[n] + 1;
            table = new int[levels][n];

            System.arraycopy(arr, 0, table[0], 0, n);

            for (int k = 1; k < levels; k++) {
                int len = 1 << k;
                int half = len >> 1;

                for (int i = 0; i + len <= n; i++) {
                    table[k][i] = Math.max(
                            table[k - 1][i],
                            table[k - 1][i + half]
                    );
                }
            }
        }

        int query(int left, int right) {
            if (left > right) {
                return 0;
            }

            int length = right - left + 1;
            int k = log[length];

            return Math.max(
                    table[k][left],
                    table[k][right - (1 << k) + 1]
            );
        }
    }

    public static void main(String[] args) throws Exception {

        BufferedReader br =
                new BufferedReader(
                        new InputStreamReader(System.in)
                );

        int n = Integer.parseInt(br.readLine().trim());
        String s = br.readLine().trim();

        /*
         * We need every rotation.
         *
         * s + s contains every rotation as a substring.
         */
        String doubled = s + s;

        /*
         * Transform:
         *
         * abc -> #a#b#c#
         *
         * This allows odd and even palindromes
         * to be handled uniformly.
         */
        int len = doubled.length();

        char[] transformed = new char[2 * len + 1];

        for (int i = 0; i < transformed.length; i++) {
            if ((i & 1) == 0) {
                transformed[i] = '#';
            } else {
                transformed[i] = doubled.charAt(i / 2);
            }
        }

        /*
         * Find palindrome radius around every center.
         */
        manacher(transformed);

        SparseTable st = new SparseTable(radius);

        StringBuilder out = new StringBuilder();

        /*
         * For each rotation:
         *
         * Original window:
         * [rotation, rotation + n - 1]
         *
         * In transformed string:
         * [2*rotation + 1,
         *  2*(rotation+n-1) + 1]
         */
        for (int rotation = 0; rotation < n; rotation++) {

            int left = 2 * rotation + 1;
            int right = 2 * (rotation + n - 1) + 1;

            int low = 1;
            int high = n;
            int answer = 1;

            /*
             * Binary search the maximum palindrome length.
             */
            while (low <= high) {

                int mid = (low + high) >>> 1;

                /*
                 * A palindrome of length mid must have
                 * its center inside this interval.
                 */
                int centerLeft = left + mid - 1;
                int centerRight = right - mid + 1;

                if (centerLeft <= centerRight &&
                    st.query(centerLeft, centerRight) >= mid) {

                    answer = mid;
                    low = mid + 1;

                } else {
                    high = mid - 1;
                }
            }

            out.append(answer).append('\n');
        }

        System.out.print(out);
    }
}
